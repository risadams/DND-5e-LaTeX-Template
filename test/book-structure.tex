% Printed-book structure (#7): chapters on right-hand pages, clean blank
% pages, art facing a chapter, section breaks and the title page

\begin{document}

\frontmatter

  \DndCredit{design}{A. Writer}
}
\tableofcontents           % page iii, recto

\mainmatter

% Page 1 ends on a recto, so a clean blank verso comes before chapter 2
\chapter{One Page}
Text.

\chapter{Blank Page Before}
Text.

% Chapter 2 ends on a recto (page 3), so the art goes on the verso (page 4)
